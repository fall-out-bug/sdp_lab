# 00-106-05: Integration test — end-to-end harness с реальным LLM

Feature: F106 (agentloop Integration)

## Goal

Написать integration test который доказывает что вся цепочка работает: `ServeBridge.DispatchAndRun` → `Harness` → реальный LLM call → tool execution → gate evaluation → evidence накоплена. Это единственное доказательство что F106 работает в production.

## Scope Files

- Требует: `OPENROUTER_API_KEY` env var; `go test -tags integration`

- [ ] **Основной тест** использует `httptest.Server` (не реальный API) — детерминированный, не требует API key, работает в CI

## Implementation Notes

Основной тест использует `httptest.Server` для детерминизма:

```go
//go:build integration

package executor_test

import (
    "context"
    "encoding/json"
    "net/http"
    "net/http/httptest"
    "os"
    "path/filepath"
    "testing"
    "time"
    
    "sdp_dev/internal/executor"
    "sdp_dev/internal/control"
    "sdp_dev/internal/agentloop"
)

func TestServeBridgeHarnessE2E(t *testing.T) {
    // Fake LLM that returns a simple text response
    srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        w.Header().Set("Content-Type", "application/json")
        json.NewEncoder(w).Encode(map[string]any{
            "choices": []map[string]any{{
                "message": map[string]any{
                    "role":    "assistant",
                    "content": "Hello world function written.",
                },
                "finish_reason": "stop",
            }},
        })
    }))
    defer srv.Close()

    tmpDir := t.TempDir()
    store := control.NewTempStore(tmpDir)
    
    // ServeBridge configured with httptest endpoint (not real OpenRouter)
    bridge := executor.NewServeBridgeWithEndpoint(store, tmpDir, srv.URL, "test-api-key")
    
    cardID := "test-f106-e2e"
    card := &control.FeatureCard{
        ID:               cardID,
        NormalizedIntent: "Write a hello world function in Go",
    }
    store.SaveCard("test-project", card)
    
    ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
    defer cancel()
    
    result, err := bridge.DispatchAndRun(ctx, "test-project", cardID)
    if err != nil {
        t.Fatalf("DispatchAndRun failed: %v", err)
    }
    if result.Status == "failed" {
        t.Fatalf("Status failed: %s", result.Error)
    }
    
    dbPath := filepath.Join(tmpDir, ".sdp", "sessions", cardID+".db")
    if _, err := os.Stat(dbPath); os.IsNotExist(err) {
        t.Fatal("session db not created")
    }
    t.Logf("Result: %+v", result)
}
```
