## Goal

TaskReconciler watches kubeopencode Task CRDs and calls the adapter layer components on each status change.

## Reconcile Logic

```
func (r *TaskReconciler) Reconcile(ctx, req) (Result, error):
  task := get Task from cache
  
  switch task.Status.Phase:
    case Pending:
      issueID := task.Labels["beads.issue"]
      lock := RunLockManager.TryAcquire(issueID, task.Name)
      if !lock: return (skip)
      gate := PolicyGate.PreDispatchModelAllowlist(task.Spec.AgentRef.Model)
      if !gate.Passed: set Task annotation "sdp.deny" = gate.Reason; return
      Beads.Claim(issueID)
      
    case Running:
      TraceEmitter.Heartbeat(task.Name)
      
    case Succeeded/Completed:
      evidence := EvidenceProjector.ProjectFromIntent(...)
      quality := QualityPipeline.Run(...)
      FSM.Transition(issueID, "review")
      
    case Failed:
      fsmState, reason := LifecycleReconciler.ReconcilePhase(current, Failed, task.Status.TerminalReason)
      FSM.Transition(issueID, fsmState)
      RunLockManager.Release(issueID)
```
